API reference

// Create a chart
const chart = StrataChart.create('#myChart', {
  type:  'bar' | 'line' | 'pie' | 'scatter',
  view:  '2d' | '3d',     // default '2d'
  theme: 'auto' | 'light' | 'dark',
  data: [
    { label: 'Jan', value: 120, category: 'Sales' },
    { label: 'Feb', value: 95,  category: 'Sales' },
  ],
  colors: ['#0d6efd', '#198754', '#dc3545'],

  // Feature flags (all default false)
  gridView:                false,
  showAxisLabels:          true,
  showScale:               true,
  showGridLabels:          false,
  highlightGridOnInteract: true,

  // Callbacks
  onReady:  (chart) => console.log('ready'),
  onChange: (view)  => console.log('view changed', view),
  onClick:  (point) => console.log('clicked', point),
})

// Instance methods
chart.toggleView()                    // toggle between 2D and 3D
chart.setView('3d')                   // set specific view
chart.update(newData)                 // replace all data
chart.addDataPoint({ label, value })  // add one point
chart.removeDataPoint(index)          // remove by index
chart.addDataPoints(points)           // add multiple
chart.removeDataPoints([0, 2])        // remove multiple by index
chart.updateDataPoint(1, newData)     // update one by index
chart.destroy()                       // unmount and clean up

// Destroy all charts at once
StrataChart.destroyAll()

Events

document.addEventListener('st:chart:ready',   e => { /* e.detail.chart, e.detail.view */ })
document.addEventListener('st:chart:change',  e => { /* e.detail.from, e.detail.to */ })
document.addEventListener('st:chart:update',  e => { /* e.detail.chart */ })
document.addEventListener('st:chart:click',   e => { /* e.detail.label, value, category, index */ })
document.addEventListener('st:chart:destroy', e => { /* e.detail.chart */ })
