API reference

// Initialise — auto-discover all [data-st-skeleton="true"] on page
SkeletonLoader.init()

// Initialise specific elements
SkeletonLoader.init('.card')
SkeletonLoader.init('#my-id')

// Show / reveal
SkeletonLoader.show('.card')
SkeletonLoader.reveal('.card')

// Staggered reveal — 150ms delay between each
SkeletonLoader.reveal('.card', { stagger: 150 })

// Reveal with callback after last element
SkeletonLoader.reveal('.card', { onReveal: () => console.log('done') })

// Toggle between skeleton and revealed
SkeletonLoader.toggle('.card')

// Reveal one element by index
SkeletonLoader.revealAt('.card', 0)

// Check if currently shimmering
SkeletonLoader.isSkeleton(el)   // → true | false

Customise via CSS variables

:root {
  --st-skeleton-base:     #e2e8f0;  /* shimmer base colour */
  --st-skeleton-shine:    #f8fafc;  /* shimmer highlight */
  --st-skeleton-duration: 1.5s;     /* animation speed */
  --st-skeleton-radius:   4px;      /* bar corner rounding */
}
