API reference

// Open by CSS selector
StrataModal.open('#myModal')

// Open by element reference
const el = document.querySelector('#myModal')
StrataModal.open(el)

// Close the current open modal
StrataModal.close()

// Events fired on document
document.addEventListener('st:modal:open',  e => { /* e.detail.modal */ })
document.addEventListener('st:modal:close', e => { /* e.detail.modal */ })

HTML attributes

<!-- Trigger: opens the modal matching data-st-target -->
<button data-st-toggle="modal" data-st-target="#myModal">Open</button>

<!-- Dismiss: closes the current open modal -->
<button data-st-dismiss="modal">Close</button>

<!-- Static backdrop: clicking outside shakes, not closes -->
<div class="modal" data-st-backdrop="static">
